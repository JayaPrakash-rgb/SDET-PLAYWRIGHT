---> setTimeout(()=>{console.log("Hello");},2000);
---> delay: The time to wait in milliseconds (1000 ms = 1sec))

// Asynchronous,promises

promise --> (resolve, reject)
  Pending   → Initial state
  Fulfilled → Operation completed successfully  -----> resolve("Data received")
  Rejected  → Operation failed------------------>  Rejected("error)
 

// example 

function getData() {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            resolve("Data received");
        }, 2000);
    });
}
 
getData().then((data) => {
    console.log(data);
}).catch((error) => {
    console.error("Error:", error);
});

 --->HANDLING PROMISES
 ---> .then() → Executes when Promise is fulfilled
---> .catch() → Executes when Promise is rejected
